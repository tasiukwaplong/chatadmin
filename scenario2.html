-- Calculate Period Totals for September 2026 SELECT SUM(CASE WHEN type = 'INCOME' THEN amount ELSE 0 END) AS total_income, SUM(CASE WHEN type = 'EXPENSE' THEN amount ELSE 0 END) AS total_expenses, (SELECT COALESCE(SUM(amount_due), 0) FROM debts_receivables WHERE status = 'UNPAID') AS total_uncollected_debts, (SUM(CASE WHEN type = 'INCOME' THEN amount ELSE 0 END) - SUM(CASE WHEN type = 'EXPENSE' THEN amount ELSE 0 END)) AS net_profit FROM transactions WHERE transaction_date BETWEEN '2026-09-01' AND '2026-09-20' AND company_id = '02_INNOVATION_LAB';
